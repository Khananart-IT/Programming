# เรียน C# EP2 การเช็คเงื่อนไข (If - Else), การ loop (for loop, while loop)
## บทที่ 6 การเช็คเงื่อนไข (If - Else)
### 1.โครงสร้างการใช้ if-else จะเป็นดังนี้

ถ้า...

    if (condition) { ... }

หรือถ้า...

    else if (condition) { ... }

ถ้าไม่เข้าเงื่อนไขไหนเลย...

    else { ... }

**ข้อสังเกต: ใน C# เงื่อนไขต้อง ใส่วงเล็บ ( ) เสมอ และคำสั่งที่จะให้ทำต้อง ใส่ไว้ในปีกกา { }**

### 2.เครื่องหมายเปรียบเทียบที่ใช้บ่อย (Comparison Operators)

**$\color{orange}{\text{==}}$** เท่ากับ (เช็คว่าเท่ากันไหม), **$\color{orange}{\text{!=}}$** ไม่เท่ากับ

**$\color{orange}{\text{>}}$** มากกว่า, **$\color{orange}{\text{<}}$** น้อยกว่า

**$\color{orange}{\text{>=}}$** มากกว่าหรือเท่ากับ, **$\color{orange}{\text{<=}}$** น้อยกว่าหรือเท่ากับ

### ตัวอย่างการใช้งานใน C#

    using System;

    class Program
    {
        static void Main()
        {
            Console.Write("Enter your score: ");
            int score = int.Parse(Console.ReadLine());

            if (score >= 80)
            {
                Console.WriteLine("Rank: S (Master!)");
            }
            else if (score >= 50)
            {
                Console.WriteLine("Rank: A (Passed)");
            }
            else
            {
                Console.WriteLine("Rank: F (Try Again!)");
            }
        }
    }

### 3.ตัวเชื่อมเงื่อนไข (Logical Operators)

-ในกรณีที่ต้องการเช็คหลายเงื่อนไขพร้อมกัน

และ (AND): ใช้ **&&** (ต้องจริงทั้งคู่) 

### ตัวอย่างการใช้

    if (hp > 0 && stamina > 0) 

หรือ (OR): ใช้ **||** (จริงแค่อันใดอันหนึ่งก็พอ) 

### ตัวอย่างการใช้

    if (hasKey == true || isVIP == true)

## บทที่ 7 การ loop

 ### 1.loop for (นับจำนวนรอบที่แน่นอน)

 loop for นิยมใช้บ่อยที่สุดสามารถรู้ได้ว่าจะให้ทำกี่รอบ

 ### โครงสร้างของ for ใน C#

    for (จุดเริ่มต้น; เงื่อนไข; การปรับค่า)

    {

        // คำสั่งที่ต้องการให้ทำซ้ำ

    }

### ตัวอย่างการนับ 1 ถึง 5 โดยใช้ for loop:

    for (int i = 1; i <= 5; i++)

    {

        Console.WriteLine("Round: " + i);

    }

ความหมายของส่วนประกอบ:

int i = 1; --> สร้างตัวนับชื่อ i เริ่มต้นที่ 1

i <= 5; --> ทำซ้ำตราบใดที่ i ยังน้อยกว่าหรือเท่ากับ 5

i++ --> ทำเสร็จแต่ละรอบ ให้เพิ่มค่า i ขึ้นทีละ 1 (เหมือน i += 1 หรือ i = i + 1)

### 2.loop while (ทำไปเรื่อยๆ จนกว่าเงื่อนไขจะเท็จ)

-ลูป while เหมาะกับงานที่เราไม่รู้จำนวนรอบที่แน่นอน เช่น ให้ผู้เล่นเล่นเกมไปเรื่อยๆ จนกว่า HP จะเหลือ 0

### โครงสร้างของ while ใน C#

    while (เงื่อนไข)

    {

        // คำสั่งที่ต้องการให้ทำซ้ำ

    }

### ตัวอย่างการใช้ while loop:

    int hp = 3;

    while (hp > 0)

    {
        Console.WriteLine("Fighting... HP left: " + hp);

        hp--; // ลดค่า hp ทีละ 1 (เท่ากับ hp = hp - 1)

    }

    Console.WriteLine("Game Over!");

## บทที่ 8 อาร์เรย์ (Arrays) และ ลิสต์ (Lists)

ปกติถ้าเรามีตัวแปรธรรมดา 1 ตัว มันเก็บค่าได้แค่ 1 ค่า เหมือนกล่องเล็กๆ ใส่ของได้ชิ้นเดียว:

    int hp = 100; // เก็บได้แค่เลขเดียว

แต่ถ้าต้องการเก็บตัวแปรให้สะดวกขึ้น ลดการใช้บรรทัดเพื่อไม่ให้โค้ดรกและเขียนยาก! เราเลยต้องใช้ตัวแปรชุดอย่าง Array หรือ List มาช่วย

### 1.Array (อาร์เรย์) — กล่องล็อกขนาด fixed

แนวคิด: เหมือน ถาดใส่ไข่ ที่มีจำนวนช่องแน่ชัดตั้งแต่แรก เช่น ถาด 6 ช่อง คุณจะแอบเอาไข่ฟองที่ 7 ยัดใส่เพิ่มเข้าไปไม่ได้!

จุดเด่น: ทำงานเร็วมาก ประหยัดพื้นที่ความจำ

ข้อจำกัด: สร้างแล้ว เพิ่ม/ลด ขนาดไม่ได้

**$\color{red}{\text{การนับ Index (ดัชนี): ในทางโปรแกรมมิ่งเราจะเริ่มนับช่องแรกที่ 0 เสมอ (0, 1, 2, 3...)}}$**

ตัวอย่างการเขียน Array:

    // การสร้าง Array เก็บข้อความ 3 ช่อง

    string[] weapons = new string[3]; 

    // การใส่ข้อมูลลงในแต่ละช่อง (Index 0 ถึง 2)

    weapons[0] = "Sword";

    weapons[1] = "Bow";

    weapons[2] = "Staff";

    // หรือจะเขียนย่อๆ แบบกำหนดค่าทันทีเลยก็ได้:

    string[] potions = { "Red Potion", "Blue Potion", "Green Potion" };

    // การเรียกใช้ข้อมูล:

    Console.WriteLine(potions[0]); // จะพิมพ์ออกมาว่า "Red Potion"

    Console.WriteLine(potions[1]); // จะพิมพ์ออกมาว่า "Blue Potion"

### 2.List (ลิสต์) — กระเป๋าขยายขนาดได้

แนวคิด: เหมือน กระเป๋าเดินทางผ้ายืด ตอนแรกกระเป๋าว่างเปล่า พอเราหยิบของใส่ มันขยายช่องเพิ่มเองอัตโนมัติ พอเอาของออก มันก็หดลงเอง!

จุดเด่น: ยืดหยุ่นมาก จะ **$\color{yellow}{\text{.Add()}}$** เพิ่ม หรือ **$\color{yellow}{\text{.Remove()()}}$**
ลบของกี่ชิ้นเมื่อไหร่ก็ได้

**$\color{red}{\text{ข้อควรจำ: ต้องเขียน using System.Collections.Generic; ไว้บรรทัดบนสุดของไฟล์เสมอ}}$**

### ตัวอย่างการเขียน List:

    using System;

    using System.Collections.Generic; // สำคัญมาก! ถ้าไม่มีบรรทัดนี้ จะใช้ List ไม่ได้

    class Program

    {
        static void Main()

        {
            // 1. สร้าง List ว่างๆ สำหรับเก็บ string

            List<string> inventory = new List<string>();

            // 2. การเพิ่มของเข้า List ใช้คำสั่ง .Add()

            inventory.Add("Health Potion"); // อยู่ Index 0

            inventory.Add("Magic Key");     // อยู่ Index 1

            inventory.Add("Dragon Shield"); // อยู่ Index 2

            // 3. การเช็คจำนวนของทั้งหมดใน List ใช้ .Count

            Console.WriteLine("Items in bag: " + inventory.Count); // ผลลัพธ์: 3

            // 4. การลบของออกจาก List ใช้คำสั่ง .Remove()

            inventory.Remove("Magic Key"); // ใช้กุญแจไปแล้ว ลบออก!

            // ตอนนี้ Magic Key หายไปแล้ว "Dragon Shield" จะเลื่อนขึ้นมาเป็น Index 1 แทนอัตโนมัติ!

            Console.WriteLine("New Item at Index 1: " + inventory[1]); // ผลลัพธ์: Dragon Shield

        }

    }
