## บทที่ 6 การเช็คเงื่อนไข (If - Else)
### 1.โครงสร้างการใช้ if-else จะเป็นดังนี้

ถ้า... (if)

    if (condition) { ... }

หรือถ้า... (else if)

    else if (condition) { ... }

ถ้าไม่เข้าเงื่อนไขไหนเลย... (else)

    else { ... }

**ข้อสังเกต: ใน C# เงื่อนไขต้อง ใส่วงเล็บ ( ) เสมอ และคำสั่งที่จะให้ทำต้อง ใส่ไว้ในปีกกา { }**

### 2.เครื่องหมายเปรียบเทียบที่ใช้บ่อย (Comparison Operators)

**$\color{orange}{\text{==}}$** เท่ากับ (เช็คว่าเท่ากันไหม), **$\color{orange}{\text{!=}}$** ไม่เท่ากับ

**$\color{orange}{\text{>}}$** มากกว่า, **$\color{orange}{\text{<}}$** น้อยกว่า

**$\color{orange}{\text{>=}}$** มากกว่าหรือเท่ากับ, **$\color{orange}{\text{<=}}$** น้อยกว่าหรือเท่ากับ

### ตัวอย่างการใช้งานใน C#

    using System;

    class Program
    {
        static void Main()
        {
            Console.Write("Enter your score: ");
            int score = int.Parse(Console.ReadLine());

            if (score >= 80)
            {
                Console.WriteLine("Rank: S (Master!)");
            }
            else if (score >= 50)
            {
                Console.WriteLine("Rank: A (Passed)");
            }
            else
            {
                Console.WriteLine("Rank: F (Try Again!)");
            }
        }
    }

### 3.ตัวเชื่อมเงื่อนไข (Logical Operators)

-ในกรณีที่ต้องการเช็คหลายเงื่อนไขพร้อมกัน

และ (AND): ใช้ **&&** (ต้องจริงทั้งคู่) 

### ตัวอย่างการใช้

    if (hp > 0 && stamina > 0) 

หรือ (OR): ใช้ **||** (จริงแค่อันใดอันหนึ่งก็พอ) 

### ตัวอย่างการใช้

    if (hasKey == true || isVIP == true)

## บทที่ 7 การ loop

 ### 1.loop for (นับจำนวนรอบที่แน่นอน)

 loop for นิยมใช้บ่อยที่สุดสามารถรู้ได้ว่าจะให้ทำกี่รอบ

 ### โครงสร้างของ for ใน C#

    for (จุดเริ่มต้น; เงื่อนไข; การปรับค่า)
    {
        // คำสั่งที่ต้องการให้ทำซ้ำ
    }

### ตัวอย่างการนับ 1 ถึง 5 โดยใช้ for loop:

    for (int i = 1; i <= 5; i++)
    {
        Console.WriteLine("Round: " + i);
    }

ความหมายของส่วนประกอบ:

int i = 1; --> สร้างตัวนับชื่อ i เริ่มต้นที่ 1

i <= 5; --> ทำซ้ำตราบใดที่ i ยังน้อยกว่าหรือเท่ากับ 5

i++ --> ทำเสร็จแต่ละรอบ ให้เพิ่มค่า i ขึ้นทีละ 1 (เหมือน i += 1 หรือ i = i + 1)

### 2.loop while (ทำไปเรื่อยๆ จนกว่าเงื่อนไขจะเท็จ)

-ลูป while เหมาะกับงานที่เราไม่รู้จำนวนรอบที่แน่นอน เช่น ให้ผู้เล่นเล่นเกมไปเรื่อยๆ จนกว่า HP จะเหลือ 0

### โครงสร้างของ while ใน C#

    while (เงื่อนไข)
    {
        // คำสั่งที่ต้องการให้ทำซ้ำ
    }

### ตัวอย่างการใช้ while loop:

    int hp = 3;
    while (hp > 0)
    {
        Console.WriteLine("Fighting... HP left: " + hp);
        hp--; // ลดค่า hp ทีละ 1 (เท่ากับ hp = hp - 1)
    }
    Console.WriteLine("Game Over!");

## บทที่ 8 อาร์เรย์ (Arrays) และ ลิสต์ (Lists)

ปกติถ้าเรามีตัวแปรธรรมดา 1 ตัว มันเก็บค่าได้แค่ 1 ค่า เหมือนกล่องเล็กๆ ใส่ของได้ชิ้นเดียว:

    int hp = 100; // เก็บได้แค่เลขเดียว

แต่ถ้าต้องการตัวกล่องที่ใหญ่ขึ้น เราไม่จำเป็นต้องเขียนโค้ดหลายบรรทัดเพื่อให้โค้ดรก เราสามารถใช้ตัวแปรชุดอย่าง Array หรือ List มาช่วยได้

### 1.Array (อาร์เรย์) — กล่องล็อกขนาด fixed

แนวคิด: เหมือนถาดใส่ไข่ที่มีจำนวนช่องแน่ชัดตั้งแต่แรก เช่น ถาด 6 ช่อง คุณจะแอบเอาไข่ฟองที่ 7 ยัดใส่เพิ่มเข้าไปไม่ได้!

จุดเด่น: ทำงานเร็วมาก ประหยัดพื้นที่ความจำ

ข้อจำกัด: สร้างแล้ว เพิ่ม/ลด ขนาดไม่ได้

**$\color{red}{\text{การนับ Index (ดัชนี): ในทางโปรแกรมมิ่งเราจะเริ่มนับช่องแรกที่ 0 เสมอ (0, 1, 2, 3...)}}$**

ตัวอย่างการเขียน Array:

    // การสร้าง Array เก็บข้อความ 3 ช่อง
    string[] weapons = new string[3];

    // การใส่ข้อมูลลงในแต่ละช่อง (Index 0 ถึง 2)
    weapons[0] = "Sword";
    weapons[1] = "Bow";
    weapons[2] = "Staff";

    // หรือจะเขียนย่อๆ แบบกำหนดค่าทันทีเลยก็ได้:
    string[] potions = { "Red Potion", "Blue Potion", "Green Potion" };

    // การเรียกใช้ข้อมูล:
    Console.WriteLine(potions[0]); // จะพิมพ์ออกมาว่า "Red Potion"
    Console.WriteLine(potions[1]); // จะพิมพ์ออกมาว่า "Blue Potion"

### 2.List (ลิสต์) — กระเป๋าขยายขนาดได้

แนวคิด: เหมือนกระเป๋าเดินทางผ้ายืด ตอนแรกกระเป๋าว่างเปล่า พอเราหยิบของใส่ มันขยายช่องเพิ่มเองอัตโนมัติ พอเอาของออก มันก็หดลงเอง!

จุดเด่น: ยืดหยุ่นมาก จะ **$\color{yellow}{\text{.Add()}}$** เพิ่ม หรือ **$\color{yellow}{\text{.Remove()}}$**
ลบของกี่ชิ้นเมื่อไหร่ก็ได้

**$\color{red}{\text{ข้อควรจำ: ต้องเขียน using System.Collections.Generic; ไว้บรรทัดบนสุดของไฟล์เสมอ}}$**

### ตัวอย่างการเขียน List:

    using System;
    using System.Collections.Generic; // สำคัญมาก! ถ้าไม่มีบรรทัดนี้ จะใช้ List ไม่ได้

    class Program
    {
        static void Main()
        {
            // 1. สร้าง List ว่างๆ สำหรับเก็บ string
            List<string> inventory = new List<string>();

            // 2. การเพิ่มของเข้า List ใช้คำสั่ง .Add()
            inventory.Add("Health Potion"); // อยู่ Index 0
            inventory.Add("Magic Key");     // อยู่ Index 1
            inventory.Add("Dragon Shield"); // อยู่ Index 2

            // 3. การเช็คจำนวนของทั้งหมดใน List ใช้ .Count
            Console.WriteLine("Items in bag: " + inventory.Count); // ผลลัพธ์: 3

            // 4. การลบของออกจาก List ใช้คำสั่ง .Remove()
            inventory.Remove("Magic Key"); // ใช้กุญแจไปแล้ว ลบออก!

            // ตอนนี้ Magic Key หายไปแล้ว "Dragon Shield" จะเลื่อนขึ้นมาเป็น Index 1 แทนอัตโนมัติ!
            Console.WriteLine("New Item at Index 1: " + inventory[1]); // ผลลัพธ์: Dragon Shield
        }
    }
    
## บทที่ 9 ลูป foreach (for - each)

เวลาเราเก็บของไว้ใน Array หรือ List การจะเอาของทุกชิ้นออกมาแสดงทีละอัน ถ้าใช้ลูป for แบบเดิม เราต้องเขียนนับ

for (int i = 0; i < inventory.Count; i++) ซึ่งแอบพิมพ์ยาวและสับสนเลข Index

C# เลยมีลูป foreach มา โดยจะเดินไปหยิบของจากกระเป๋าทีละชิ้นๆ ตั้งแต่ชิ้นแรกจนถึงชิ้นสุดท้ายให้เองอัตโนมัติ

โครงสร้างของ foreach:

    foreach ([ประเภทข้อมูล] [ชื่อตัวแปรชั่วคราว] in [ชื่อ Array หรือ List])
    {
        // ทำอะไรกับตัวแปรชั่วคราวนั้น
    }

ตัวอย่างการใช้งาน foreach:

    using System;
    using System.Collections.Generic;

    class Program
    {
        static void Main()
        {
            List<string> skillList = new List<string>();
            skillList.Add("Fireball");
            skillList.Add("Ice Blast");
            skillList.Add("Heal");

            Console.WriteLine("=== Skill Book ===");

            // อ่านว่า: สำหรับ skill (string) แต่ละอันที่อยู่ใน skillList
            foreach (string skill in skillList)
            {
                Console.WriteLine("- Learned: " + skill);
            }
        }
    }

ผลลัพธ์บนหน้าจอ:

    === Skill Book ===
    - Learned: Fireball
    - Learned: Ice Blast
    - Learned: Heal

## บทที่ 10 การ return ค่า
### กฎเหล็ก 2 ข้อของการ Return ค่าใน C#
### 1.ประเภทข้อมูลต้องตรงกันเป๊ะ!ถ้าหนูประกาศหน้าชื่อ Method ว่าจะส่งคืนค่าประเภทไหนคำสั่ง return ข้างใน ต้องคืนค่าประเภทนั้นออกมาเสมอ

เช่นกำหนดไว้เป็น int --> ต้อง return ตัวเลขจำนวนเต็ม

กำหนดไว้เป็น string --> ต้อง return ข้อความ

กำหนดไว้เป็น bool --> ต้อง return true หรือ false

กำหนด Method ไว้เป็น void ---> ห้ามคืนค่าอะไรเลย (ไม่มีคำสั่ง return หรือเขียนแค่ return; เปล่าๆ เพื่อเด้งออกจากฟังก์ชัน)

### 2.คำสั่ง return คือการ "จบงานทันที"
-เมื่อไหร่ก็ตามที่โค้ดรันมาเจอคำสั่ง return มันจะ เด้งออกจาก Method นั้นทันที โค้ดบรรทัดที่อยู่ใต้ return ลงไปจะไม่ถูกทำงานอีกเลย

### เปรียบเทียบโค้ดชัดๆ 4 รูปแบบ
รูปแบบที่ 1: แบบ void (ไม่มีการ Return)

    using System;

    class Program
    {
        // ใช้ void แปลว่า "ทำอย่างเดียว ไม่ส่งค่ากลับ"
        static void SayHello(string name)
        {
            Console.WriteLine("Hello " + name + "!");
            // ไม่ต้องมีคำสั่ง return
        }

        static void Main()
        {
            // เรียกใช้เฉยๆ ไม่ต้องสร้างตัวแปรมารับค่า
            SayHello("Alex"); 
        }
    }

รูปแบบที่ 2: แบบมี Return ค่า (ส่งค่ากลับมาใช้งานต่อ)

คราวนี้จะเปลี่ยนคำว่า void เป็น int (ชนิดข้อมูลที่จะส่งกลับ)

    using System;

    class Program
    {
        // กำหนด int ไว้หน้าชื่อ Method แปลว่า "บวกเลขเสร็จแล้ว ต้องส่ง int กลับไปน้า"
        static int AddNumbers(int a, int b)
        {
            int result = a + b;
            return result; //ส่งค่าเลขผลลัพธ์กลับไปให้คนที่เรียก!
        }

        static void Main()
        {
            // เรียกใช้ AddNumbers(5, 10) 
            // พอมันส่งค่า 15 กลับมา เราก็เอาตัวแปร sum มารับค่านั้นไว้!
            int sum = AddNumbers(5, 10);
        
            Console.WriteLine("Total: " + sum); // พิมพ์ 15
        
            // หรือเอาไปคำนวณต่อได้ทันที!
            int doubleSum = sum * 2; 
            Console.WriteLine("Double Total: " + doubleSum); // พิมพ์ 30
        }
    }

รูปแบบที่ 3: Return ค่าแบบ bool (เช็คเงื่อนไข แล้วตอบว่า "จริง" หรือ "เท็จ")

ถ้าเป็นในเกมจะใช้บ่อย เช่น เช็คว่าผู้เล่นมีเงินพอซื้อของไหม?

    using System;

    class Program
    {
        // ส่งค่ากลับเป็น bool (true / false)
        static bool CanBuyItem(int playerMoney, int itemPrice)
        {
            if (playerMoney >= itemPrice)
            {
                return true;  // ถ้าเงินพอ ส่ง true กลับไป แล้วจบฟังก์ชันทันที!
            }
            else
            {
                return false; // ถ้าเงินไม่พอ ส่ง false กลับไป
            }
        }

        static void Main()
        {
            int myMoney = 50;
            int swordPrice = 100;

            // เรียกฟังก์ชันเช็ค แล้วเอาผลลัพธ์ (true/false) ไปใส่ใน if
            if (CanBuyItem(myMoney, swordPrice) == true)
            {
                Console.WriteLine("You bought the sword!");
            }
            else
            {
                Console.WriteLine("You don't have enough money!");
            }
        }
    }

รูปแบบที่ 4: ตัวอย่างคำสั่ง return แบบหยุดการทำงานทันที (Exit Early)

บางครั้งเราใช้ return ใน void เพื่อ "ยกเลิกการทำงานกลางคัน" เวลาเกิดเหตุการณ์ที่ไม่ถูกต้อง เช่น พลังชีวิตหมดแล้ว ห้ามเดินต่อ!

    static void PlayerMove(int hp)
    {
        // ถ้า HP เหลือ 0 หรือติดลบ ให้เลิกทำทันที!
        if (hp <= 0)
        {
            Console.WriteLine("Player is dead! Cannot move.");
            return; // เจอคำสั่งนี้จะเด้งออกจาก Method ทันที บรรทัดข้างล่างจะไม่ทำ!
        }

        // ถ้า HP มากกว่า 0 ถึงจะทำงานบรรทัดนี้
        Console.WriteLine("Player moved forward!");
    }
