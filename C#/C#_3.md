โดยจะประกอบไปด้วย:

### 1.เครื่องหมายคำนวณแบบผสม (Compound Assignment Operators)
ใช้เมื่อเราต้องการนำตัวแปรเดิมมาคำนวณแล้วเก็บค่ากลับเข้าตัวแปรเดิม

|ตัวเต็ม|ตัวย่อ|การทำงาน|
| -------- | -------- | -------- |
| score = score + 10; | score += 10; | บวกเพิ่ม 10 จากค่าเดิม |
| hp = hp - 25; | hp -= 25; | ลดลง 25 จากค่าเดิม |
| damage = damage * 2; | damage *= 2; | คูณ 2 จากค่าเดิม (เช่น สถานะ Critical) |
| gold = gold / 2; | gold /= 2; | หาร 2 จากค่าเดิม (เช่น โดนปล้นเงินหายครึ่งนึง) |
| count = count % 3; | count %= 3; | หาเศษจากการหารแล้วเก็บค่าเดิม |

### ตัวอย่าง:

    int playerHp = 100;
    playerHp -= 30; // HP เหลือ 70 ทันที

### 2.การเพิ่ม / ลดค่าทีละ 1 (Increment & Decrement)
-เมื่อต้องการเพิ่มหรือลดค่าตัวเลขแค่ 1 (นิยมมากในลูป for หรือการนับจำนวน)

|การเขียนแบบเต็ม|การเขียนแบบตัวย่อ|การทำงาน|
| -------- | -------- | -------- |
| count = count + 1; | count++; | Increment (เพิ่มทีละ 1) |
| count = count - 1; | count--; | Decrement (ลดทีละ 1) |
| damage = damage * 2; | damage *= 2; | คูณ 2 จากค่าเดิม (เช่น สถานะ Critical) |
| gold = gold / 2; | gold /= 2; | หาร 2 จากค่าเดิม (เช่น โดนปล้นเงินหายครึ่งนึง) |
| count = count % 3; | count %= 3; | หาเศษจากการหารแล้วเก็บค่าเดิม |

**$\color{red}{\text{ข้อสังเกตเพิ่มเติม (เกร็ดระดับสูง):}}$**

**i++ (Post-increment): นำค่าเดิมไปใช้ก่อน แล้วค่อย +1 ทีหลัง**
**++i (Pre-increment): +1 ก่อน แล้วค่อยนำค่าใหม่ไปใช้**

### 3.ทางลัด if-else บรรทัดเดียว (Ternary Operator)
-เหมาะกับงานที่ต้องการเช็คเงื่อนไขง่ายๆ เพื่อ เลือกคืนค่า 1 ใน 2 อย่างใส่ตัวแปร 
โครงสร้าง:

$$\text{ตัวแปร} = (\text{เงื่อนไข}) \text{ ? } \text{ถ้าจริงเอาค่านี่} \text{ : } \text{ถ้าเท็จเอาค่านี่};$$

แบบเต็ม:C#string status;
if (hp > 0) {
    status = "Alive";
} else {
    status = "Dead";
}
ย่อเหลือบรรทัดเดียว:C#string status = (hp > 0) ? "Alive" : "Dead";
4. ทางลัดต่อข้อความ (String Interpolation)เวลาเอาข้อความมาต่อกับตัวแปร การใช้เครื่องหมาย + หลายๆ ตัวจะทำให้สับสนและลายตา C# จึงมีระบบ $ ช่วยให้พิมพ์ตัวแปรซ้อนในข้อความได้เลยแบบเดิม (ใช้ + ต่อสาย):C#Console.WriteLine("Player " + name + " has " + hp + " HP.");
แบบย่อมืออาชีพ (ใส่ $ ไว้หน้าเครื่องหมายอัญประกาศ):C#Console.WriteLine($"Player {name} has {hp} HP.");
(ใส่ชื่อตัวแปรไว้ในปีกกา { } ได้เลย สะอาดตาขึ้นมาก!)5. ทางลัดเช็คค่าว่าง null (Null-Coalescing Operator)ใช้กำหนด "ค่าสำรอง" ในกรณีที่ตัวแปรอาจจะไม่มีข้อมูล (เป็น null)เครื่องหมาย ??C#// ถ้า playerName เป็น null ให้ใช้ชื่อสำรองว่า "Guest" แทน
string displayName = playerName ?? "Guest";
