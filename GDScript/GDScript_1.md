# เรียน GDScript EP1 ตัวแปร, ฟังก์ชันและวงจรชีวิตของเกม (Functions & Lifecycle), การสร้างเงื่อนไขและการตัดสินใจ (If-Else & Operators), การรับปุ่มกดและการเคลื่อนที่ (Input & Movement)
## บทที่ 1 ตัวแปร
ในบทนี้เราจะมาทำความรู้จักกับสิ่งที่เรียกว่า ตัวแปร (Variables) เปรียบเสมือน "กล่องวิเศษ" ที่คอมพิวเตอร์ใช้เก็บจำข้อมูลต่าง ๆ ของเกม เช่น คะแนน, ชื่อผู้เล่น, หรือจำนวนไอเทม

### 1.วิธีประกาศตัวแปร (Declaring Variables)
ในภาษา GDScript เราจะสร้างกล่องข้อมูลขึ้นมาด้วยการพิมพ์คำว่า var นำหน้าเสมอ เช่น:

    var player_name = "Mark"
    var gold = 500

### 2.ชนิดของข้อมูลที่ต้องใช้บ่อย (Data Types)
เพื่อให้เกมของเราทำงานได้เร็วและไม่มีบั๊ก ใน Godot 4 เรานิยมใส่ Type Hint (การระบุประเภทของกล่อง) ด้วยการใส่เครื่องหมายโคลอน (:) ตามด้วยชื่อชนิดข้อมูล ซึ่งมี 4 ตัวหลัก ๆ ดังนี้:

1.String (ข้อความ): ใช้เก็บตัวอักษร ชื่อ หรือคำพูด ยล้อมรอบด้วยเครื่องหมายคำพูด " "

    var item_name: String = "Wooden Sword"

2.int (Integer - เลขจำนวนเต็ม): ใช้เก็บตัวเลขที่ไม่มีจุดทศนิยม เหมาะกับพวกเลเวล, เลือด, หรือจำนวนเงิน

    var level: int = 1

3.float (Floating-point - เลขทศนิยม): ใช้เก็บตัวเลขที่มีจุดทศนิยม เหมาะกับความเร็ว หรือพิกัดตำแหน่ง

    var speed: float = 150.5

4.bool (Boolean - ความจริง): เก็บค่าได้แค่ 2 อย่างเท่านั้นคือ true (จริง) หรือ false (เท็จ) เหมาะกับการเช็กสถานะ เช่น เกมโอเวอร์หรือยัง? เปิดประตูหรือยัง?

    var is_game_over: bool = false

## 3. ตัวแปรพิเศษสำหรับคนออกแบบ: @export

ตัวนี้เด็ดมากค๊าบ! ถ้าใส่ @export ไว้หน้าตัวแปรตัวไหน ตัวแปรนั้นจะไปโผล่เป็นช่องให้สามารถคลิกกรอกตัวเลขหรือเลือกไฟล์ในหน้าต่าง Inspector (ฝั่งขวา) ได้ทันทีโดยไม่ต้องเปิดโค้ดแก้บ่อย ๆ!

    @export var max_stamina: int = 50  # ช่องนี้จะโผล่ใน Inspector ให้ปรับเล่นได้เลย!

## บทที่ 2 — ฟังก์ชันและวงจรชีวิตของเกม (Functions & Lifecycle)
ถ้า ตัวแปร คือกล่องเก็บข้อมูล ฟังก์ชัน (Function) ก็คือ "ชุดคำสั่งหรือการกระทำ" ที่เราสั่งให้เกมทำงานค๊าบ โดยเราจะใช้คำย่อว่า func นำหน้าเสมอ

ซึ่งใน Godot จะมีฟังก์ชันพิเศษที่ระบบเตรียมไว้ให้เรา ซึ่งมันจะทำงานตาม "ช่วงเวลา" ของเกม (เราเรียกว่าวงจรชีวิต หรือ Lifecycle) มี 3 ตัวหลัก ๆ ที่หนูต้องสนิทกับมันไว้:

### 1.func _ready(): (ช่วงเวลาเกิด)
ฟังก์ชันนี้จะทำงานแค่  **$\color{red}{\text{ครั้งเดียวเท่านั้น!}}$** ตอนที่ Node นั้น ๆ ถูกโหลดขึ้นมาในฉากเสร็จปุ๊บ เหมาะมากสำหรับตั้งค่าเริ่มต้นของเกม

**ตัวอย่างการใช้: สั่งให้เลือดตัวละครเต็มหลอดตอนเริ่มเกม, สั่งให้ตู้เกมสุ่มแจกตั๋วฟรีเมื่อผู้เล่นเปิดฉากเข้ามา**

### 2.func _process(delta): (ทำงานตลอดเวลา ทุกเฟรมเรต)
ฟังก์ชันนี้จะรัน  **$\color{red}{\text{ตลอดเวลา ทุก ๆ เฟรมของเกม}}$** (ถ้าเกมวิ่ง 60 FPS มันก็จะรัน 60 ครั้งใน 1 วินาที!)

**ตัวอย่างการใช้: เช็กว่าผู้เล่นกดปุ่มเดินหรือยัง, สั่งนับเวลาถอยหลังของตู้เกม, เช็กตลอดเวลาว่าผู้เล่นกดปุ่มเปลี่ยนฉากหรือยัง**

💡 ตัวแปร delta คือ เวลาที่ใช้ไปในแต่ละเฟรม เอาไว้ช่วยคำนวณให้เกมรันสมูท

### 3.func _physics_process(delta): (ทำงานตลอดเวลา สายฟิสิกส์)
คล้ายกับ _process มาก แต่ถูกออกแบบมาให้รันด้วย ความเร็วที่คงที่แน่นอนเสมอ ไม่ขึ้นอยู่กับความกระตุกของหน้าจอ

**ตัวอย่างการใช้: ทุกครั้งที่หนูเขียนโค้ดคุมตัวละครวิ่ง, กระโดด, ตกตึก หรือชนตู้เกม ต้องเขียนในนี้เท่านั้น เพื่อป้องกันไม่ให้ตัวละครเดินทะลุกำแพงเวลาเกมกระตุก**

### 🛠️ ตัวอย่างการเขียนฟังก์ชันร่วมกันในเกมจริง:

    extends CharacterBody2D

    var arcade_tokens: int = 0  # ตัวแปรจากบทที่ 1

    # 🟢 ทำงานครั้งเดียวตอนเริ่มเกม
    func _ready():
        arcade_tokens = 10     # เริ่มเกมมา แจกเหรียญฟรีให้ผู้เล่น 10 เหรียญ
        print("Welcome to my arcade, Your coins: ", arcade_tokens)

    # 🔄 ทำงานตลอดเวลาเพื่อเช็กปุ่มกด
    func _process(delta):
        # สมมติว่าถ้าผู้เล่นกดปุ่ม "หยอดเหรียญ" (ขอมองข้ามโค้ดปุ่มกดไปก่อนน้า)
        # เราสามารถสร้างฟังก์ชันขึ้นมาเองเพื่อสั่งงานได้ เช่น:
        pass

    # ⚙️ ฟังก์ชันที่เราสร้างขึ้นมาเอง (Custom Function)
    func insert_token():
        if arcade_tokens > 0:
            arcade_tokens -= 1  # หักเหรียญออก 1 เหรียญ
            print("Coin use!, Coin remain: ", arcade_tokens)
        else:
            print("Your coin is out...")

## บทที่ 3 — การสร้างเงื่อนไขและการตัดสินใจ (If-Else & Operators)
ถ้าเกมไม่มีการตัดสินใจ มันก็จะไม่สนุกใช่ไหมล่ะ? เช่น ถ้าเราหยอดเหรียญไม่ครบ ตู้เกมต้องไม่ยอมให้เล่น หรือถ้าคะแนนถึงเป้า ตู้เกมต้องแจกตั๋วรางวัล! ลอจิกพวกนี้เราจะใช้คำสั่ง if, elif, และ else ในการควบคุม

### 1.โครงสร้างการตัดสินใจ
if (ถ้า): เงื่อนไขแรกที่เราต้องการเช็ก

elif (หรือถ้า): (ย่อมาจาก Else If) เอาไว้เช็กเงื่อนไขถัดไป ในกรณีที่เงื่อนไขแรกไม่เป็นจริง

else (นอกเหนือจากนั้น): ถ้าเงื่อนไขข้างบนทั้งหมดไม่เป็นจริงเลย ให้มาทำที่ท่อนนี้ค๊าบ

    if score >= 1000:
        print("Great! you get GOLDEN TICKET!! 🎫✨")
    elif score >= 500:
        print("Nice! You get silver ticket! 🎫")
    else:
        print("Try again! 👾")

**$\color{red}{\text{(⚠️ ข้อควรระวัง: หลังเครื่องหมาย : บรรทัดต่อไปต้องกด Tab ย่อหน้าเข้าไปเสมอเหมือนภาษา Python เกมถึงจะรู้ว่าโค้ดนั้นเป็นลูกของเงื่อนไขไหน)}}$**

### 2.เครื่องหมายเปรียบเทียบ (Comparison Operators)
เวลาเราจะเปรียบเทียบตัวเลขใน if เราจะใช้เครื่องหมายเหล่านี้:

(=)  เท่ากับ (ใช้เครื่องหมายเท่ากับ 2 ตัวติดกันน้า)

(!=)  ไม่เท่ากับ

(> , <)  มากกว่า , น้อยกว่า

(>= , <=) มากกว่าหรือเท่ากับ , น้อยกว่าหรือเท่ากับ

### 3.ตัวเชื่อมเงื่อนไข and กับ or (เพิ่มความซับซ้อนให้สมองกล)
and (และ): เงื่อนไขรอบข้างต้องเป็นจริงทั้งหมด ถึงจะผ่านเข้าประตูได้

    # ตัวอย่าง: จะเล่นตู้เกมได้ ต้องมีเหรียญ และ ตู้เกมต้องไม่พัง
    if tokens > 0 and is_machine_broken == false:
        print("Start the game!")

or (หรือ): เงื่อนไขใดเงื่อนไขหนึ่งเป็นจริงแค่ข้อเดียว ก็ยอมให้ผ่านแล้ว

    # ตัวอย่าง: เกมจะจบลง ถ้าเวลาหมด หรือ เลือดเหลือ 0
    if time_left <= 0 or health <= 0:
        print("Game Over!")

## บทที่ 4 — การรับปุ่มกดและการเคลื่อนที่ (Input & Movement)
ในเกม 2D เราขับเคลื่อนวัตถุด้วยระบบพิกัด แกน X (ซ้าย-ขวา) และ แกน Y (บน-ลงล่าง) ค๊าบ (ทบทวนความจำ: แกน Y ในลอจิกเกม 2D ยิ่งค่าเป็นบวกจะยิ่งวิ่งลงข้างล่าง)

ใน Godot 4 การจะทำให้ตัวละครเดินได้ มี 2 กลไกสำคัญที่ต้องทำงานร่วมกัน:

### 1.การเช็กปุ่มกด (Input)
Godot มีปุ่มมาตรฐานที่ตั้งค่ามาให้เราตั้งแต่เกิด (เรียกว่า UI Actions) เราสามารถใช้คำสั่ง **Input.is_action_pressed()** มาเช็กได้ตลอดเวลาใน _physics_process:

"ui_left" = ปุ่มลูกศรซ้าย หรือ ปุ่ม A

"ui_right" = ปุ่มลูกศรขวา หรือ ปุ่ม D

"ui_up" = ปุ่มลูกศรบน หรือ ปุ่ม W

"ui_down" = ปุ่มลูกศรล่าง หรือ ปุ่ม S

### 2.ตัวแปรอัจฉริยะ velocity และคำสั่ง move_and_slide()
ถ้าใช้ Node ประเภท CharacterBody2D ทำตัวละคร จะได้ของแถมสุดโกงมา 2 อย่างทันที:

velocity (ความเร็ว): เป็นตัวแปรประเภท Vector2 (มีทั้งค่า X และ Y ในตัว) เอาไว้บอกเกมว่า "เฮ้! ตอนนี้ฉันอยากให้ตัวละครวิ่งไปทางไหนด้วยความเร็วเท่าไหร่"

move_and_slide(): ฟังก์ชันสำเร็จรูป สั่งบรรทัดเดียวปุ๊บ ตัวละครจะขยับเดินตามค่า velocity ทันที แถมช่วยคำนวณระบบชนกำแพงให้แบบสไลด์นุ่ม ๆ ไม่ติดขัด

🛠️ ตัวอย่างสคริปต์เดิน 4 ทิศทางแบบเกม RPG / ร้านอาเขต
ลองมาดูโค้ดประกอบร่างจริง:

    extends CharacterBody2D

    @export var speed: float = 200.0  # ความเร็วในการเดิน (ปรับได้ใน Inspector)

    func _physics_process(delta):
        # 📦 1. เคลียร์ความเร็วให้เป็น 0 ก่อนในทุกๆ เฟรม (ถ้าไม่กดปุ่ม ตัวละครต้องหยุดเดิน)
        var direction: Vector2 = Vector2.ZERO
        
        # 🕹️ 2. เช็กปุ่มกดเพื่อหาทิศทาง (Direction)
        if Input.is_action_pressed("ui_right"):
            direction.x += 1
        if Input.is_action_pressed("ui_left"):
            direction.x -= 1
        if Input.is_action_pressed("ui_down"):
            direction.y += 1
        if Input.is_action_pressed("ui_up"):
            direction.y -= 1
            
        # 🎯 3. คำนวณความเร็ว (เอาทิศทางมาคูณกับความเร็วที่เราตั้งไว้)
        # ใช้คำสั่ง .normalized() เพื่อป้องกันไม่ให้ตัวละครเดินทะแยงมุมเร็วกว่าปกติค๊าบ
        velocity = direction.normalized() * speed
        
        # 🏃‍♂️ 4. สั่งให้ตัวละครเคลื่อนที่จริงบนหน้าจอ
        move_and_slide()
